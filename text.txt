<script>
async function generateWorkout() {
    const muscles = document.getElementById("muscles").value.trim();
    const goal = document.getElementById("goal").value;
    const days = document.getElementById("days").value;
    const equipment = document.getElementById("equipment").value.trim();

    if (!muscles) {
        alert("Please enter at least one muscle group.");
        return;
    }

    const workoutPlanDiv = document.getElementById("workoutPlan");
    workoutPlanDiv.innerHTML = "<p>Generating workout... please wait.</p>";

    const prompt = `
Create a ${days}-day workout plan for someone whose goal is ${goal}, targeting the following muscles: ${muscles}.
Include exercises (using ${equipment || "any equipment"}), sets, reps, rest times, and short descriptions.
Format it clearly for display and printing.
    `;

    try {
        const response = await axios.post('https://api.openai.com/v1/chat/completions', {
            model: "gpt-5-mini",
            messages: [{ role: "user", content: prompt }],
            temperature: 0.7
        }, {
            headers: {
                'Authorization': 'Bearer YOUR_OPENAI_API_KEY',
                'Content-Type': 'application/json'
            }
        });

        const aiText = response.data.choices[0].message.content;
        workoutPlanDiv.innerHTML = aiText;

    } catch (error) {
        console.error(error);
        workoutPlanDiv.innerHTML = "<p>Error generating workout. Check console for details.</p>";
    }
}

function printWorkout() {
    const printContents = document.getElementById("workoutPlan").innerHTML;
    const originalContents = document.body.innerHTML;
    document.body.innerHTML = printContents;
    window.print();
    document.body.innerHTML = originalContents;
}
</script>
